          <div class="rc-meta"><i class="fa-solid fa-store"></i> ${d.rest}</div>
        </div>
        <div class="rc-stars">${starsHTML(r.rating)}<span class="when">${r.date}</span></div>
      </div>
      <h4 class="rc-title">${r.title}</h4>
      <p class="rc-text">${r.text}</p>
      ${r.tags.length?`<div class="rc-tags">${r.tags.map(t=>`<span class="rc-tag"><i class="fa-solid fa-utensils"></i>${t}</span>`).join('')}</div>`:''}
      <div class="rc-foot">
        <div class="reviewer">
          <span class="avatar" style="background:${GRADS[r.id%GRADS.length]}">${initials(r.name)}</span>
          ${r.name}
          ${r.verified?'<span class="v-badge"><i class="fa-solid fa-circle-check"></i> Verified order</span>':'<span class="v-badge new"><i class="fa-solid fa-sparkles"></i> New</span>'}
        </div>
        <div style="display:flex;align-items:center;gap:14px">
          <button class="helpful-btn ${isLiked?'liked':''}" data-id="${r.id}">
            <i class="fa-${isLiked?'solid':'regular'} fa-thumbs-up"></i> Helpful \u00b7 ${h}
          </button>
          <button class="report-link" onclick="toast('Thanks! Our food moderators will check it.')"><i class="fa-regular fa-flag"></i> Report</button>
        </div>
      </div>
    </article>`;
  }).join('');
  $('#loadMore').style.display=visible>=list.length?'none':'inline-flex';
}
