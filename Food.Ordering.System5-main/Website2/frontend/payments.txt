async function apiFetch(endpoint, options = {}){
  const ctrl = new AbortController();
  const timer = setTimeout(() => ctrl.abort(), 2200);
  try{
    const user = getUser();
    const res = await fetch(API_BASE_URL + endpoint, {
      ...options,
      headers:{ 'Content-Type':'application/json', ...(user && user.token ? {'Authorization':'Bearer '+user.token} : {}), ...(options.headers||{}) },
      signal: ctrl.signal
    });
    clearTimeout(timer);
    if(res.status === 401){ store.del('foody_user'); location.href='login.html'; throw new Error('Session expired'); }
    if(!res.ok) throw new Error('Request failed');
    return await res.json();
  }catch(e){ clearTimeout(timer); return null; }
}

/* ---------------- PAYMENT LOGIC ---------------- */
let checkoutData = null;
let currentView = 'summary'; // summary | processing | success | failure
let generatedOrderId = null;

function cartLines(){
  return getCart().map(ci => { const f = foodById(ci.id); return f ? {f, qty:ci.qty} : null; }).filter(Boolean);
}
function computeSummary(){
  const lines = cartLines();
  const subtotal = lines.reduce((s,l) => s + effPrice(l.f)*l.qty, 0);
  let discount = 0, fee = subtotal > 0 ? (subtotal >= FREE_DELIVERY_ABOVE ? 0 : BASE_DELIVERY) : 0;
  let coupon = getCoupon();
  if(coupon){
    const c = COUPONS[coupon];
    if(!c || subtotal < c.min){ coupon = null; store.del('foody_coupon'); }
    else{
      if(c.type==='percent') discount = Math.min(Math.round(subtotal*c.value/100), c.max);
      else if(c.type==='delivery') fee = 0;
    }
  }
  const total = Math.max(0, subtotal - discount) + fee;
  return { lines, subtotal, discount, fee, total, coupon };
}

  const s = computeSummary();
  const box = document.getElementById('sumItems');

      </div>
      <span class="si-total">₹${effPrice(l.f)*l.qty}</span>

async function processPayment(){
  currentView = 'processing';
  renderActionArea();

  // Simulate API call to backend / payment gateway
  await apiFetch('/orders', { method:'POST', body: JSON.stringify({}) }) || await demo(true, 2200);

  // 85% success rate for demo online, 100% for COD
  const isCOD = checkoutData.method === 'cod';
  const success = isCOD ? true : (Math.random() < 0.85);

  if(success){
    generatedOrderId = 'FD' + Date.now().toString().slice(-6);
    const s = computeSummary();
    const addr = getAddresses().find(a => a.id === checkoutData.address_id) || {};
    
    // Save Order
    const order = {
      id: generatedOrderId,
      date: new Date().toISOString(),
      items: s.lines.map(l => ({id:l.f.id, name:l.f.name, price:effPrice(l.f), qty:l.qty, image:l.f.image, restaurant_id:l.f.restaurant_id})),
      restaurant_names: [...new Set(s.lines.map(l => REST_NAMES[l.f.restaurant_id] || 'Restaurant'))].join(', '),
      address: addr,
      subtotal: s.subtotal,
      discount: s.discount,
      fee: s.fee,
      total: s.total,
      coupon: s.coupon,
      payment_method: checkoutData.method,
      payment_status: isCOD ? 'COD' : 'PAID',
      status: 'CONFIRMED'
    };
    const orders = getOrders();
    orders.unshift(order);
    saveOrders(orders);

    // Clear Cart & Checkout Data
    saveCart([]);
    store.del('foody_coupon');
    store.del('foody_checkout');
    updateCartBadge();

    currentView = 'success';
  } else {
    currentView = 'failure';
  }
  renderActionArea();
}

function triggerConfetti(){
  const wrap = document.getElementById('confettiWrap');
  wrap.innerHTML = '';
  const colors = ['#FFC107', '#FF9800', '#2E7D32', '#D32F2F', '#1976D2'];
  for(let i=0; i<60; i++){
    const c = document.createElement('div');
    c.className = 'confetti';
    c.style.left = Math.random() * 100 + '%';
    c.style.background = colors[Math.floor(Math.random() * colors.length)];
    c.style.width = (Math.random() * 8 + 5) + 'px';
    c.style.height = (Math.random() * 8 + 5) + 'px';
    c.style.animationDuration = (Math.random() * 2 + 1.5) + 's';
    c.style.animationDelay = (Math.random() * 0.5) + 's';
    wrap.appendChild(c);
  }
  setTimeout(() => wrap.innerHTML = '', 4000);
}
